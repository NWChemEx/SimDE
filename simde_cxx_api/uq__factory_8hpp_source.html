54inline UQKind uq_kind_from_string(const std::string& uq_type) {
55 if(uq_type == "none") return UQKind::none;
56 if(uq_type == "uncertain") return UQKind::uncertain;
57 if(uq_type == "interval") return UQKind::interval;
58 if(uq_type == "affine") return UQKind::affine;
59 if(uq_type == "thresholded affine") return UQKind::thresholded_affine;
60 if(uq_type == "triplex") return UQKind::triplex;
61 throw std::runtime_error(
62 "simde::uq_kind_from_string: Invalid UQ type name " + uq_type);
63}

66inline std::string to_string(UQKind kind) {

212class UQFactory {
213public:
216 UQFactory() : m_impl_(std::make_shared<detail::UncertainFactoryImpl>()) {}
217
219 explicit UQFactory(UQKind kind) : m_impl_(make_impl_(kind)) {}
220
222 wtf::fp::Float operator()(double center, double radius) const {
223 return (*m_impl_)(center, radius);
224 }
225
228 wtf::fp::Float operator()(double nominal, double minus, double plus) const {
229 return (*m_impl_)(nominal, minus, plus);
230 }
231
233 UQKind kind() const noexcept { return m_impl_->kind(); }
234
236 bool operator==(const UQFactory& rhs) const noexcept {
237 return m_impl_->equal(*rhs.m_impl_);
238 }
239
241 bool operator<(const UQFactory& rhs) const noexcept {
242 return m_impl_->less(*rhs.m_impl_);
243 }
244
245private:
246 static std::shared_ptr<const UQFactoryBase> make_impl_(UQKind kind) {
247 switch(kind) {
248 case UQKind::none:
249 return std::make_shared<detail::NoneFactoryImpl>();
250 case UQKind::uncertain:
251 return std::make_shared<detail::UncertainFactoryImpl>();
252 case UQKind::interval:
253 return std::make_shared<detail::IntervalFactoryImpl>();
254 case UQKind::affine:
255 return std::make_shared<detail::AffineFactoryImpl>();
256 case UQKind::thresholded_affine:
257 return std::make_shared<detail::ThresholdedAffineFactoryImpl>();
258 case UQKind::triplex:
259 return std::make_shared<detail::TriplexFactoryImpl>();
260 }
261 throw std::runtime_error("simde::UQFactory: Unrecognized UQKind");
262 }
263
264 std::shared_ptr<const UQFactoryBase> m_impl_;
265};

296template<typename FloatType = double, typename VisitorType>
297decltype(auto) visit_uq_kind(UQKind kind, VisitorType&& visitor) {
298 using namespace tensorwrapper::types;
299 switch(kind) {
300 case UQKind::none: return visitor(std::type_identity<FloatType>{});
301 case UQKind::uncertain:
302 return visitor(std::type_identity<uncertain_type<FloatType>>{});
303 case UQKind::interval:
304 return visitor(std::type_identity<interval_type<FloatType>>{});
305 case UQKind::affine:
306 return visitor(std::type_identity<affine_type<FloatType>>{});
307 case UQKind::thresholded_affine:
308 return visitor(
309 std::type_identity<thresholded_affine_type<FloatType>>{});
310 case UQKind::triplex:
311 return visitor(std::type_identity<triplex_type<FloatType>>{});
312 }
313 throw std::runtime_error("simde::visit_uq_kind: Unrecognized UQKind");
314}
