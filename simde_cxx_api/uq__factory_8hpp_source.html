47inline UQKind uq_kind_from_string(const std::string& uq_type) {
48 if(uq_type == "none") return UQKind::none;
49 if(uq_type == "uncertain") return UQKind::uncertain;
50 if(uq_type == "interval") return UQKind::interval;

175class UQFactory {
176public:
179 UQFactory() : m_impl_(std::make_shared<detail::UncertainFactoryImpl>()) {}
180
182 explicit UQFactory(UQKind kind) : m_impl_(make_impl_(kind)) {}
183
185 wtf::fp::Float operator()(double center, double radius) const {
186 return (*m_impl_)(center, radius);
187 }
188
190 UQKind kind() const noexcept { return m_impl_->kind(); }
191
193 bool operator==(const UQFactory& rhs) const noexcept {
194 return m_impl_->equal(*rhs.m_impl_);
195 }
196
198 bool operator<(const UQFactory& rhs) const noexcept {
199 return m_impl_->less(*rhs.m_impl_);
200 }
201
202private:
203 static std::shared_ptr<const UQFactoryBase> make_impl_(UQKind kind) {
204 switch(kind) {
205 case UQKind::none:
206 return std::make_shared<detail::NoneFactoryImpl>();
207 case UQKind::uncertain:
208 return std::make_shared<detail::UncertainFactoryImpl>();
209 case UQKind::interval:
210 return std::make_shared<detail::IntervalFactoryImpl>();
211 case UQKind::affine:
212 return std::make_shared<detail::AffineFactoryImpl>();
213 case UQKind::thresholded_affine:
214 return std::make_shared<detail::ThresholdedAffineFactoryImpl>();
215 }
216 throw std::runtime_error("simde::UQFactory: Unrecognized UQKind");
217 }
218
219 std::shared_ptr<const UQFactoryBase> m_impl_;
220};

251template<typename FloatType = double, typename VisitorType>
252decltype(auto) visit_uq_kind(UQKind kind, VisitorType&& visitor) {
253 using namespace tensorwrapper::types;
254 switch(kind) {
255 case UQKind::none: return visitor(std::type_identity<FloatType>{});
256 case UQKind::uncertain:
257 return visitor(std::type_identity<uncertain_type<FloatType>>{});
258 case UQKind::interval:
259 return visitor(std::type_identity<interval_type<FloatType>>{});
260 case UQKind::affine:
261 return visitor(std::type_identity<affine_type<FloatType>>{});
262 case UQKind::thresholded_affine:
263 return visitor(
264 std::type_identity<thresholded_affine_type<FloatType>>{});
265 }
266 throw std::runtime_error("simde::visit_uq_kind: Unrecognized UQKind");
267}
